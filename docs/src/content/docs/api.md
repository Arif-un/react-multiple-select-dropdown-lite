## Events

| Prop             | Called                                                                                                                                                                               |
| ---------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------ |
| `onChange`       | after any change, with the new values and their options                                                                                                                              |
| `onMenuOpen`     | when the menu opens; with `open`, to ask to open it                                                                                                                                  |
| `onMenuClose`    | when the menu closes (outside click, Escape, Tab, a pick, scrolled out of view); with `open`, to ask to close it                                                                     |
| `onSearchChange` | `(text, reason)`: the search field's text as it changes, raw. `reason` is `'input'` (typed or pasted), `'pick'` (a pick cleared it) or `'close'` (the menu closed). Each change once |
| `onClear`        | after the clear button empties the value, `fixed` values aside                                                                                                                       |

## Types

```ts
interface MultiSelectOption {
  label: string // searched, and the default display
  value: string
  disabled?: boolean
  fixed?: boolean // once selected, it can't be removed
  style?: React.CSSProperties
  classes?: string // extra class names
  [key: string]: any // your own fields, returned in onChange's options
}

// with getLabel and getValue, options can be any shape T, inferred from `options`
interface MultiSelectGroup<T = MultiSelectOption> {
  type: 'group' | 'submenu' // a submenu's title opens childs in a popover beside it
  title: ReactNode
  childs: (T | MultiSelectGroup<T>)[] | MultiSelectLoader<T> // a loader loads them the first time they show
  emptyDataLabel?: ReactNode // shown when childs is empty
  header?: ReactNode // first in its rows, a row with renderList: text, not an option (no role, the highlight passes it)
  footer?: ReactNode // last in its rows, as header
  collapsible?: boolean // 'group' only: the title shows and hides childs
  collapsed?: boolean // with collapsible, childs start hidden
  loading?: boolean // as the dropdown's loading, onLoadMore and loadError, for an array of childs you load
  onLoadMore?: () => void
  loadError?: unknown
  loadMoreLabel?: ReactNode // its own label for the button at the end of childs
  style?: React.CSSProperties // on its title row
  classes?: string // extra class names, on its title row
  [key: string]: any // your own fields, read back in renderGroupTitle's group
}

// loadOptions, or a group's childs: returns, or resolves to, the options, or { options, next } while there are more
type MultiSelectLoader<T = MultiSelectOption> = (
  request: MultiSelectLoadRequest
) => MultiSelectLoaded<T> | PromiseLike<MultiSelectLoaded<T>>

interface MultiSelectLoadRequest {
  search: string // the search text; '' for a group outside what loadOptions loaded, or onSearch's options
  next?: any // the next the list's last part came with: undefined for the first
  signal: AbortSignal // aborts once another search takes its place, or the dropdown unmounts
}

type MultiSelectLoaded<T> =
  | (T | MultiSelectGroup<T>)[]
  | { options: (T | MultiSelectGroup<T>)[]; next?: unknown } // a truthy next: more to load

// getLabel's second argument
interface MultiSelectLabelMeta {
  context: 'menu' | 'value' // an option's row, or a chip (customChip's label) and the single value
  search: string // the search text for menu rows; '' for values, and when search reads the label's text
  selected: boolean // picked: always for values; false when search reads the label's text
}

// what renderList gets
interface MultiSelectListProps {
  count: number // rows of the menu or an open submenu: options, titles and empty groups' labels, groups flattened
  row: (index: number) => ReactNode // renders one row; nothing past the end
  activeIndex: number // the keyboard highlight's row, -1 for none
}

// what renderGroupTitle gets. C: customValue is on. To annotate it without customValue, pass false,
// MultiSelectGroupTitleProps<Person, false>, or group's options are typed Person | { label, value }
interface MultiSelectGroupTitleProps<
  T = MultiSelectOption,
  C extends boolean = boolean
> {
  group: MultiSelectGroup<T | { label: string; value: string }> // as in options; { label, value } for customValue's group of created values, unless C is false. A copy for a group with a loader in it, its loaders' childs those loaded
  title: ReactNode // the group's title
  total: number // options in it, nested groups' and submenus' too, disabled ones and those hideSelected hides included; a loader's, those loaded
  selected: number // how many of those are picked
  open: boolean // a collapsible group shows its childs, a submenu is open; true for a plain group
  icon: ReactNode // the .msl-expand-icn span, to place or leave out; null for a plain group
  toggle: () => void // selects its enabled options, or once none is left to pick deselects them but fixed ones
}

// what menuHeader and menuFooter get as functions
interface MultiSelectMenuProps<
  T = MultiSelectOption,
  C extends boolean = boolean
> {
  search: string // the search text
  options: (T | { label: string; value: string })[] // the enabled options the menu lists, flat; { label, value } for created values, unless C is false
  selected: number // how many of them are selected
  selectAll: () => void // selects those not selected yet, up to limit (one in single select)
  deselectAll: () => void // deselects them but fixed ones
}
```
